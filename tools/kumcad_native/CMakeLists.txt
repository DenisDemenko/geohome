cmake_minimum_required(VERSION 3.21)
project(GeoHomeKumCadInterop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(LCAD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LCAD_ENABLE_DWG OFF CACHE BOOL "" FORCE)
set(LCAD_ENABLE_LASZIP OFF CACHE BOOL "" FORCE)
set(LCAD_ENABLE_3D ON CACHE BOOL "" FORCE)
if(NOT DEFINED KUMCAD_ROOT)
  message(FATAL_ERROR "Set KUMCAD_ROOT to the pinned KumCAD source checkout")
endif()
add_subdirectory("${KUMCAD_ROOT}" upstream_kumcad EXCLUDE_FROM_ALL)
if(NOT TARGET lcad_core)
  message(FATAL_ERROR "KumCAD core target not available")
endif()
target_include_directories(lcad_core PUBLIC "${KUMCAD_ROOT}/src")
add_executable(geohome_kumcad_step_smoke smoke.cpp)
target_link_libraries(geohome_kumcad_step_smoke PRIVATE lcad_core)
